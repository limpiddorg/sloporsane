include(cmake/sources.cmake)
include(FetchContent)
find_package(ftxui 6 QUIET)
if(NOT ftxui_FOUND)
    set(FTXUI_BUILD_DOCS OFF CACHE BOOL "" FORCE)
    set(FTXUI_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
    set(FTXUI_BUILD_TESTS OFF CACHE BOOL "" FORCE)
    set(FTXUI_ENABLE_INSTALL OFF CACHE BOOL "" FORCE)
    FetchContent_Declare(ftxui URL https://github.com/ArthurSonzogni/FTXUI/archive/refs/tags/v6.1.9.tar.gz)
    FetchContent_MakeAvailable(ftxui)
    install(FILES "${ftxui_SOURCE_DIR}/LICENSE" DESTINATION share/sos-ftxui/licenses RENAME ftxui.txt COMPONENT ftxui)
endif()
find_package(Threads REQUIRED)
include(cmake/markdown.cmake)
add_library(sos-ftxui-core STATIC ${SOS_FTXUI_CORE_SOURCES})
target_include_directories(sos-ftxui-core PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
target_link_libraries(sos-ftxui-core PUBLIC nlohmann_json::nlohmann_json Threads::Threads sos::agent-runtime ftxui::screen ftxui::dom)
add_executable(sos-ftxui ${SOS_FTXUI_SOURCES})
target_link_libraries(sos-ftxui PRIVATE sos-ftxui-core sos-ftxui-markdown ftxui::component ftxui::dom ftxui::screen)
add_dependencies(sos-ftxui sosd)
add_custom_target(build-ftxui DEPENDS sos-ftxui)
install(TARGETS sos-ftxui sosd RUNTIME DESTINATION bin COMPONENT ftxui)
if(IS_ABSOLUTE "$ENV{XDG_DATA_HOME}")
    set(FTXUI_DEFAULT_INSTALL_DIR "$ENV{XDG_DATA_HOME}/sos/ftxui")
else()
    set(FTXUI_DEFAULT_INSTALL_DIR "$ENV{HOME}/.local/share/sos/ftxui")
endif()
set(SOS_FTXUI_INSTALL_DIR "${FTXUI_DEFAULT_INSTALL_DIR}" CACHE PATH "Directory for versioned native TUI installations")
set(SOS_FTXUI_BIN_DIR "$ENV{HOME}/.local/bin" CACHE PATH "Directory for the sos-ftxui launcher")
add_custom_target(install-ftxui
    COMMAND "${CMAKE_COMMAND}" "-DBUILD_DIR=${PROJECT_BINARY_DIR}"
        "-DAPP_DIR=${SOS_FTXUI_INSTALL_DIR}" "-DBIN_DIR=${SOS_FTXUI_BIN_DIR}"
        -P "${CMAKE_CURRENT_SOURCE_DIR}/cmake/install.cmake"
    DEPENDS sos-ftxui
    COMMENT "Installing the native sos-ftxui integration and daemon"
    VERBATIM
)
if(BUILD_TESTING)
    if(NOT TARGET GTest::gtest_main)
        find_package(GTest 1.17.0 REQUIRED)
    endif()
    add_executable(ftxui-tests ${SOS_FTXUI_TEST_SOURCES})
    target_link_libraries(ftxui-tests PRIVATE sos-ftxui-core sos-ftxui-markdown GTest::gtest_main ftxui::component ftxui::dom)
    include(GoogleTest)
    gtest_discover_tests(ftxui-tests)
    add_executable(ftxui-e2e-tests ${SOS_FTXUI_E2E_TEST_SOURCES})
    add_executable(ftxui-fixture-daemon ${SOS_FTXUI_FIXTURE_SOURCES})
    target_link_libraries(ftxui-fixture-daemon PRIVATE daemon-sessions)
    target_link_libraries(ftxui-e2e-tests PRIVATE sos-ftxui-core GTest::gtest_main)
    if(CMAKE_SYSTEM_NAME STREQUAL "Linux")
        target_link_libraries(ftxui-e2e-tests PRIVATE util)
    endif()
    target_compile_definitions(ftxui-e2e-tests PRIVATE SOS_FTXUI_BINARY="$<TARGET_FILE:sos-ftxui>"
        SOS_FTXUI_FIXTURE_DAEMON="$<TARGET_FILE:ftxui-fixture-daemon>")
    add_dependencies(ftxui-e2e-tests sos-ftxui ftxui-fixture-daemon)
    gtest_discover_tests(ftxui-e2e-tests WORKING_DIRECTORY "${PROJECT_BINARY_DIR}" PROPERTIES TIMEOUT 30)
endif()
