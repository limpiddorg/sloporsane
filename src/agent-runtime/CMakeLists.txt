include(cmake/sources.cmake)
include(cmake/tree_sitter.cmake)

find_package(Threads REQUIRED)
find_package(yaml-cpp 0.8 QUIET)
if(NOT yaml-cpp_FOUND)
    include(FetchContent)
    set(YAML_CPP_BUILD_TESTS OFF CACHE BOOL "" FORCE)
    set(YAML_CPP_BUILD_TOOLS OFF CACHE BOOL "" FORCE)
    FetchContent_Declare(yaml-cpp URL https://github.com/jbeder/yaml-cpp/archive/refs/tags/0.8.0.tar.gz)
    FetchContent_MakeAvailable(yaml-cpp)
endif()
find_package(LibXml2 REQUIRED)
find_package(OpenSSL REQUIRED COMPONENTS Crypto SSL)
include(cmake/websocket.cmake)

find_package(cpr 1.14.2 QUIET)
if(NOT cpr_FOUND)
    include(FetchContent)
    if(POLICY CMP0135)
        cmake_policy(SET CMP0135 NEW)
    endif()
    set(CPR_USE_SYSTEM_CURL ON CACHE BOOL "Use installed libcurl" FORCE)
    set(CPR_BUILD_TESTS OFF CACHE BOOL "Build CPR tests" FORCE)
    FetchContent_Declare(cpr
        URL https://github.com/libcpr/cpr/archive/refs/tags/1.14.2.tar.gz
    )
    FetchContent_MakeAvailable(cpr)
endif()

find_package(nlohmann_json 3.12.0 QUIET)
if(NOT nlohmann_json_FOUND)
    include(FetchContent)
    if(POLICY CMP0135)
        cmake_policy(SET CMP0135 NEW)
    endif()
    FetchContent_Declare(nlohmann_json
        URL https://github.com/nlohmann/json/releases/download/v3.12.0/json.tar.xz
    )
    FetchContent_MakeAvailable(nlohmann_json)
endif()

add_library(agent-runtime STATIC ${AGENT_RUNTIME_SOURCES})
add_library(sos::agent-runtime ALIAS agent-runtime)

target_include_directories(agent-runtime PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}/..")
target_compile_features(agent-runtime PUBLIC cxx_std_23)
target_link_libraries(agent-runtime PUBLIC nlohmann_json::nlohmann_json)
target_link_libraries(agent-runtime PUBLIC cpr::cpr)
target_link_libraries(agent-runtime PRIVATE Threads::Threads)
target_link_libraries(agent-runtime PRIVATE yaml-cpp::yaml-cpp)
target_link_libraries(agent-runtime PRIVATE LibXml2::LibXml2)
target_link_libraries(agent-runtime PRIVATE OpenSSL::Crypto)
target_link_libraries(agent-runtime PRIVATE ixwebsocket::ixwebsocket)
target_link_libraries(agent-runtime PRIVATE sos-tree-sitter sos-tree-sitter-bash)
